cmake_minimum_required(VERSION 3.16)
project(d1max_navigation_bt)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)

add_library(navigation_bt_core src/engine.cpp)
target_include_directories(navigation_bt_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(navigation_bt_core behaviortree_cpp_v3)
target_compile_options(navigation_bt_core PRIVATE -Wall -Wextra -Wpedantic)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/navigator_node.cpp")
  find_package(rclcpp REQUIRED)
  find_package(rclcpp_action REQUIRED)
  find_package(nav2_util REQUIRED)
  find_package(rclcpp_lifecycle REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(d1max_navigation_bt_interfaces REQUIRED)
  find_package(d1max_planning_interfaces REQUIRED)
  find_package(Eigen3 REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(ament_index_cpp REQUIRED)
  find_package(nlohmann_json REQUIRED)
  add_executable(navigator_node src/navigator_node.cpp)
  target_link_libraries(navigator_node navigation_bt_core nlohmann_json::nlohmann_json Eigen3::Eigen OpenSSL::Crypto)
  ament_target_dependencies(navigator_node rclcpp rclcpp_action nav2_util rclcpp_lifecycle std_msgs
    geometry_msgs nav_msgs d1max_navigation_bt_interfaces d1max_planning_interfaces ament_index_cpp)
  install(TARGETS navigator_node DESTINATION lib/${PROJECT_NAME})
endif()

install(TARGETS navigation_bt_core EXPORT navigation_bt_coreTargets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY trees DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_navigation_bt test/test_engine.cpp)
  target_link_libraries(test_navigation_bt navigation_bt_core)
  target_compile_definitions(test_navigation_bt PRIVATE
    DEFAULT_BT_XML="${CMAKE_CURRENT_SOURCE_DIR}/trees/navigate_with_global_relocalization.xml")
  ament_add_gtest(test_navigation_bt_runtime test/test_runtime_policy.cpp)
  target_link_libraries(test_navigation_bt_runtime navigation_bt_core)
  ament_add_gtest(test_navigation_task_contract test/test_task_contract.cpp)
  target_link_libraries(test_navigation_task_contract navigation_bt_core)
  ament_target_dependencies(test_navigation_task_contract d1max_navigation_bt_interfaces)
  ament_add_gtest(test_initial_pose_transaction test/test_initial_pose_transaction.cpp)
  target_link_libraries(test_initial_pose_transaction navigation_bt_core)
  ament_target_dependencies(test_initial_pose_transaction d1max_navigation_bt_interfaces)
  ament_add_gtest(test_execution_owner test/test_execution_owner.cpp)
  target_link_libraries(test_execution_owner navigation_bt_core)
  ament_target_dependencies(test_execution_owner d1max_planning_interfaces d1max_navigation_bt_interfaces)
  target_compile_definitions(test_execution_owner PRIVATE
    DEFAULT_BT_XML="${CMAKE_CURRENT_SOURCE_DIR}/trees/navigate_with_global_relocalization.xml")
endif()
ament_export_targets(navigation_bt_coreTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(behaviortree_cpp_v3)
ament_package()
