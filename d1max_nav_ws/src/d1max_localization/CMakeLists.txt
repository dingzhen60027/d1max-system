cmake_minimum_required(VERSION 3.8)
project(d1max_localization)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(pcl_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(fast_gicp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
ament_python_install_package(${PROJECT_NAME})
add_executable(fused_icp_matcher src/fused_icp_matcher.cpp src/rotation_deskewer.cpp)
target_include_directories(fused_icp_matcher PRIVATE include ${PCL_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIR})
target_link_libraries(fused_icp_matcher ${PCL_LIBRARIES})
ament_target_dependencies(fused_icp_matcher rclcpp sensor_msgs std_msgs geometry_msgs diagnostic_msgs pcl_ros pcl_conversions livox_ros_driver2 fast_gicp)
add_executable(dual_lidar_adapter src/dual_lidar_adapter.cpp)
target_include_directories(dual_lidar_adapter PRIVATE include)
ament_target_dependencies(dual_lidar_adapter rclcpp sensor_msgs diagnostic_msgs rosgraph_msgs livox_ros_driver2 tf2 tf2_ros tf2_geometry_msgs)
install(TARGETS fused_icp_matcher dual_lidar_adapter RUNTIME DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/icp_fusion_bridge scripts/localization_supervisor scripts/lio_localizer scripts/lio_predictor scripts/navigation_output DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md UPSTREAM.md ROBUST_LOCALIZATION.md NAVIGATION_ESTIMATION.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_navigation_estimation test/test_navigation_estimation.py TIMEOUT 30)
  ament_add_gtest(test_registration_information test/test_registration_information.cpp)
  target_include_directories(test_registration_information PRIVATE include ${EIGEN3_INCLUDE_DIR})
  ament_add_gtest(test_input_clock test/test_input_clock.cpp)
  target_include_directories(test_input_clock PRIVATE include)
  ament_add_gtest(test_fused_icp_gate test/test_fused_icp_gate.cpp)
  target_include_directories(test_fused_icp_gate PRIVATE include)
  ament_add_gtest(test_rotation_deskewer test/test_rotation_deskewer.cpp src/rotation_deskewer.cpp)
  target_include_directories(test_rotation_deskewer PRIVATE include ${EIGEN3_INCLUDE_DIR})
endif()
ament_package()
