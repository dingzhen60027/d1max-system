cmake_minimum_required(VERSION 3.8)
project(d1max_trajectory_tracker)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(bspline_opt REQUIRED)
find_package(d1max_planning_interfaces REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(trajectory_tracker src/trajectory_tracker.cpp)
target_include_directories(trajectory_tracker PRIVATE include)
ament_target_dependencies(trajectory_tracker bspline_opt d1max_planning_interfaces geometry_msgs nav_msgs rclcpp std_msgs std_srvs)
target_link_libraries(trajectory_tracker Eigen3::Eigen nlohmann_json::nlohmann_json Threads::Threads OpenSSL::Crypto)
install(TARGETS trajectory_tracker DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_fixed_bspline_sampler test/test_fixed_bspline_sampler.cpp)
  if(TARGET test_fixed_bspline_sampler)
    target_include_directories(test_fixed_bspline_sampler PRIVATE include)
    ament_target_dependencies(test_fixed_bspline_sampler bspline_opt d1max_planning_interfaces)
    target_link_libraries(test_fixed_bspline_sampler Eigen3::Eigen)
  endif()
  ament_add_gtest(test_tracker_core test/test_tracker_core.cpp)
  if(TARGET test_tracker_core)
    target_include_directories(test_tracker_core PRIVATE include)
    ament_target_dependencies(test_tracker_core bspline_opt d1max_planning_interfaces)
    target_link_libraries(test_tracker_core Eigen3::Eigen)
  endif()
  ament_add_gtest(test_tracker_v2 test/test_tracker_v2.cpp)
  if(TARGET test_tracker_v2)
    target_include_directories(test_tracker_v2 PRIVATE include)
    ament_target_dependencies(test_tracker_v2 bspline_opt d1max_planning_interfaces)
    target_link_libraries(test_tracker_v2 Eigen3::Eigen Threads::Threads)
  endif()
  ament_add_gtest(test_navigation_state_contract test/test_navigation_state_contract.cpp)
  if(TARGET test_navigation_state_contract)
    target_include_directories(test_navigation_state_contract PRIVATE include)
    ament_target_dependencies(test_navigation_state_contract bspline_opt d1max_planning_interfaces)
    target_link_libraries(test_navigation_state_contract Eigen3::Eigen)
  endif()
endif()
ament_package()
