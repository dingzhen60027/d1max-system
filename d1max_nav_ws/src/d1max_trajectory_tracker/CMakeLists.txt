cmake_minimum_required(VERSION 3.8)
project(d1max_trajectory_tracker)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(bspline_opt REQUIRED)
find_package(d1max_planning_interfaces REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(Threads REQUIRED)
add_executable(trajectory_tracker src/trajectory_tracker.cpp)
add_executable(local_follower src/local_follower_node.cpp)
target_include_directories(local_follower PRIVATE include)
ament_target_dependencies(local_follower bspline_opt d1max_planning_interfaces geometry_msgs nav_msgs rclcpp std_msgs)
target_link_libraries(local_follower Eigen3::Eigen nlohmann_json::nlohmann_json Threads::Threads)
target_include_directories(trajectory_tracker PRIVATE include)
ament_target_dependencies(trajectory_tracker bspline_opt d1max_planning_interfaces geometry_msgs nav_msgs rclcpp std_msgs std_srvs)
target_link_libraries(trajectory_tracker Eigen3::Eigen nlohmann_json::nlohmann_json Threads::Threads)
install(TARGETS trajectory_tracker local_follower DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
